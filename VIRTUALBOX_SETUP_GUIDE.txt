5. STEP 4: DISABLING SECURITY CONTROLS & INTERFERING SERVICES
================================================================================
To prevent Windows Defender or Windows Update from killing malware samples
before the behavioral agent can observe and record their execution:

1. Disable Windows Defender Real-Time & Tamper Protection:
   • Open Start Menu -> Settings -> Update & Security -> Windows Security.
   • Click "Virus & threat protection".
   • Under "Virus & threat protection settings", click "Manage settings".
   • Turn OFF:
     - Real-time protection
     - Cloud-delivered protection
     - Automatic sample submission
     - Tamper Protection

2. Disable SmartScreen:
   • In Windows Security -> "App & browser control" -> "Reputation-based protection settings".
   • Turn OFF "Check apps and files" and "SmartScreen for Microsoft Edge".

3. Disable User Account Control (UAC):
   • Press Win + R, type `UserAccountControlSettings` and press Enter.
   • Move the slider all the way down to "Never notify". Click OK.

4. Disable Screen Sleep & Lockout:
   • Go to Settings -> System -> Power & sleep.
   • Set "Screen" to Never, and "Sleep" to Never.

5. Disable Windows Updates:
   • Press Win + R, type `services.msc` and press Enter.
   • Find "Windows Update", double click, set Startup Type to "Disabled", click Stop, and click OK.

class TelemetryWriter:
    """Thread-safe, multi-path direct-flushing telemetry log writer."""
    def __init__(self, target_paths: List[str]):
        self.target_paths = target_paths
        self.lock = threading.Lock()
        self._init_log()

    def _init_log(self):
        header = (
            "================================================================================\n"
            "           THREAT SANDBOX BEHAVIORAL TELEMETRY AGENT v2.5 (SOC EDITION)        \n"
            "================================================================================\n"
            f"Session Initialized: {datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n"
            f"Target Hostname:    {socket.gethostname()}\n"
            f"User Context:       {os.environ.get('USERNAME', 'Unknown')}\n"
            f"Platform:           {sys.platform} (Win32 Telemetry Engine Active)\n"
            f"Sink Targets:       {', '.join(self.target_paths)}\n"
            "================================================================================\n\n"
        )
        for target in self.target_paths:
            try:
                os.makedirs(os.path.dirname(target), exist_ok=True)
                with open(target, "w", encoding="utf-8") as f:
                    f.write(header)
                    f.flush()
            except Exception:
                pass

    def write_line(self, line: str):
        with self.lock:
            ts = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            entry = f"[{ts}] {line}\n"
            print(entry.strip())

            for target in self.target_paths:
                try:
                    os.makedirs(os.path.dirname(target), exist_ok=True)
                    with open(target, "a", encoding="utf-8", errors="replace") as f:
                        f.write(entry)
                        f.flush()
                        os.fsync(f.fileno())
                except Exception:
                    pass

# Module 1: Process Monitor (Pure Python + Windows tasklist/wmic)
# ---------------------------------------------------------------------------
class ProcessMonitor:
    def __init__(self):
        self.baseline_pids: Set[int] = set()
        self.known_processes: Dict[int, Dict] = {}
        self.captured_tokens: Set[str] = set()
        self.threat_score = 0
        self.detected_threats = []

    def get_current_process_list(self) -> Dict[int, Dict]:
        procs = {}
        try:
            cmd = ["tasklist", "/FO", "CSV", "/NH"]
            res = subprocess.run(cmd, capture_output=True, text=True, errors="replace", timeout=3)
            if res.returncode == 0:
                reader = csv.reader(io.StringIO(res.stdout))
                for row in reader:
                    if len(row) >= 2:
                        pname = row[0].strip().strip('"').lower()
                        try:
                            pid = int(row[1].strip().strip('"'))
                            procs[pid] = {
                                "pid": pid,
                                "name": pname,
                                "cmdline": pname
                            }
                        except ValueError:
                            continue
        except Exception as e:
            writer.write_line(f"[WARN-PROC] Process query note: {e}")

        try:
            wmic_cmd = ["wmic", "process", "get", "ProcessId,CommandLine", "/FORMAT:CSV"]
            wres = subprocess.run(wmic_cmd, capture_output=True, text=True, errors="replace", timeout=3)
            if wres.returncode == 0:
                wreader = csv.reader(io.StringIO(wres.stdout))
                for wrow in wreader:
                    if len(wrow) >= 3:
                        try:
                            wcmd = wrow[1].strip()
                            wpid = int(wrow[2].strip())
                            if wpid in procs and wcmd:
                                procs[wpid]["cmdline"] = wcmd
                        except (ValueError, IndexError):
                            continue
        except Exception:
            pass

        return procs

    def build_baseline(self):
        writer.write_line("[*] Establishing baseline system process snapshot...")
        current = self.get_current_process_list()
        self.baseline_pids = set(current.keys())
        self.known_processes = current
        writer.write_line(f"[+] Baseline established: {len(self.baseline_pids)} background processes registered.")

    def inspect_new_processes(self):
        current = self.get_current_process_list()
        current_pids = set(current.keys())

        new_pids = current_pids - self.known_processes.keys()

        for pid in new_pids:
            pinfo = current[pid]
            pname = pinfo["name"]
            cmdline = pinfo.get("cmdline", pname)

            self.captured_tokens.add(pname)
            writer.write_line(f"[PROCESS-SPAWN] PID: {pid} | Image: {pname} | CommandLine: {cmdline}")

            cmd_lower = cmdline.lower()
            for pattern, (desc, severity) in SUSPICIOUS_COMMANDS.items():
                if pattern in cmd_lower:
                    self.threat_score += severity
                    threat_entry = f"{pname} -> {desc}"
                    if threat_entry not in self.detected_threats:
                        self.detected_threats.append(threat_entry)
                        self.captured_tokens.add(pattern)
                        self.captured_tokens.add(pname)
                        writer.write_line(f"[SECURITY-ALERT] High-Risk Vector: {desc} (Pattern: '{pattern}', Severity: +{severity})")

            self.known_processes[pid] = pinfo

class FileMonitor:
    def __init__(self):
        self.watch_dirs = [
            DESKTOP_DIR,
            os.environ.get("TEMP", r"C:\Windows\Temp"),
            os.path.join(USER_PROFILE, "AppData", "Local", "Temp")
        ]
        self.seen_files: Set[str] = set()

    def build_baseline(self):
        for wdir in self.watch_dirs:
            if os.path.exists(wdir):
                try:
                    for root, _, files in os.walk(wdir):
                        for f in files:
                            self.seen_files.add(os.path.join(root, f))
                except Exception:
                    pass
        writer.write_line(f"[+] Filesystem baseline registered: {len(self.seen_files)} existing files indexed.")

    def check_changes(self, captured_tokens: Set[str]):
        for wdir in self.watch_dirs:
            if not os.path.exists(wdir):
                continue
            try:
                for f in os.listdir(wdir):
                    fpath = os.path.join(wdir, f)
                    if fpath not in self.seen_files and os.path.isfile(fpath):
                        self.seen_files.add(fpath)
                        if f == LOG_FILENAME:
                            continue
                        ext = os.path.splitext(f)[1].lower()
                        writer.write_line(f"[FILE-DROP] Path: {fpath} ({os.path.getsize(fpath)} bytes)")
                        if ext in [".exe", ".bat", ".ps1", ".vbs", ".dll", ".js"]:
                            captured_tokens.add(f.lower())
                            writer.write_line(f"[FILE-ALERT] Executable/script payload deposited: {f}")
            except Exception:
                pass

class NetworkMonitor:
    def __init__(self):
        self.seen_connections: Set[str] = set()

    def check_connections(self, captured_tokens: Set[str]):
        try:
            res = subprocess.run(["netstat", "-ano"], capture_output=True, text=True, errors="replace", timeout=4)
            if res.returncode == 0:
                for line in res.stdout.splitlines():
                    line = line.strip()
                    if "ESTABLISHED" in line or "SYN_SENT" in line:
                        parts = re.split(r"\s+", line)
                        if len(parts) >= 5:
                            proto, local, remote, state, pid = parts[0], parts[1], parts[2], parts[3], parts[4]
                            if not remote.startswith("127.0.0.1") and not remote.startswith("0.0.0.0") and not remote.startswith("[::"):
                                conn_key = f"{proto}_{remote}_{pid}"
                                if conn_key not in self.seen_connections:
                                    self.seen_connections.add(conn_key)
                                    captured_tokens.add("outbound_socket")
                                    writer.write_line(f"[NETWORK-BEACON] {proto} Remote: {remote} | State: {state} | Owning PID: {pid}")
        except Exception:
            pass

# Module 4: Registry Persistence Monitor
# ---------------------------------------------------------------------------
class RegistryMonitor:
    def __init__(self):
        self.seen_run_entries: Set[str] = set()

    def check_persistence(self, captured_tokens: Set[str]):
        if sys.platform != "win32":
            return
        try:
            import winreg
            keys = [
                (winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Run"),
                (winreg.HKEY_LOCAL_MACHINE, r"Software\Microsoft\Windows\CurrentVersion\Run")
            ]
            for hkey, subkey in keys:
                try:
                    with winreg.OpenKey(hkey, subkey, 0, winreg.KEY_READ) as key:
                        count = winreg.QueryInfoKey(key)[1]
                        for i in range(count):
                            name, val, _ = winreg.EnumValue(key, i)
                            entry_sig = f"{name}={val}"
                            if entry_sig not in self.seen_run_entries:
                                self.seen_run_entries.add(entry_sig)
                                captured_tokens.add("regedit.exe")
                                captured_tokens.add("reg.exe")
                                writer.write_line(f"[REGISTRY-PERSISTENCE] Autorun modified: {name} -> {val}")
                except Exception:
                    pass
        except Exception:
            pass

    net_mon = NetworkMonitor()
    reg_mon = RegistryMonitor()

    proc_mon.build_baseline()
    file_mon.build_baseline()

    writer.write_line("[+] Sandbox environment armed. Waiting for payload detonation...")

    cycle = 0
    try:
        while True:
            cycle += 1
            proc_mon.inspect_new_processes()
            file_mon.check_changes(proc_mon.captured_tokens)
            net_mon.check_connections(proc_mon.captured_tokens)
            reg_mon.check_persistence(proc_mon.captured_tokens)

            if cycle % 5 == 0:
                write_summary(proc_mon)

            time.sleep(POLL_INTERVAL_SECS)
    except KeyboardInterrupt:
        writer.write_line("[*] Telemetry Agent shutting down gracefully.")
        write_summary(proc_mon)
    except Exception as e:
        writer.write_line(f"[CRITICAL] Telemetry Agent loop encountered error: {e}")
        write_summary(proc_mon)
